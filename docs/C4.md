```mermaid
C4Context
    title Sistema de precios de perfumes — Contexto

    Person(admin, "Administrador", "Configura tiendas, selectores y monitorea el estado del sistema.")
    Person(visitor, "Usuario Visitante", "Busca perfumes, compara precios y configura alertas personales.")

    System(perfumPrices, "perfum-prices-v2", "Obtiene, almacena y expone precios históricos de perfumes desde múltiples tiendas en Chile.")

    System_Ext(retailSites, "Tiendas Retail", "Paris, Falabella, Ripley, Lider, Fasa, SAIRAM, Multimarcasperfumes, Alisha, Silk — sitios web con precios.")
    System_Ext(mlAPI, "MercadoLibre API", "API pública gratuita. Entrega precios y datos estructurados de perfumes.")

    System_Ext(telegramBot, "Telegram Bot API", "Canal alternativo de notificaciones en tiempo real.")

    Rel(admin, perfumPrices, "Gestiona tiendas, selectores y monitorea salud", "HTTPS")
    Rel(visitor, perfumPrices, "Busca, filtra, compara precios y configura alertas", "HTTPS")
    Rel(perfumPrices, retailSites, "Extrae precios mediante scraping", "Puppeteer/HTTPS")

    Rel(perfumPrices, telegramBot, "Envía notificaciones por Telegram", "HTTPS")
    Rel(perfumPrices, supabase, "Almacena y consulta precios e historial", "PostgreSQL/HTTPS")
```

## Nivel 2 — Contenedores

Muestra los procesos ejecutables, almacenes de datos y cómo se comunican entre sí.

```mermaid
C4Container
    title perfum-prices-v2 — Contenedores

    Person(admin, "Administrador")
    Person(visitor, "Usuario Visitante")

    System_Ext(retailSites, "Tiendas Retail (scraping)")
    System_Ext(mlAPI, "MercadoLibre API")
    System_Ext(gmailSMTP, "Gmail SMTP")
    System_Ext(telegramBot, "Telegram Bot API")

    System_Boundary(perfumPrices, "perfum-prices-v2") {

        Container(frontend, "Frontend", "Angular + TypeScript", "SPA. Búsqueda, filtros, historial de precios, gráficos de tendencia, panel Admin, gestión de alertas y favoritos.")

        Container(api, "API REST", "Node.js + Express + TypeScript", "Expone endpoints para precios, productos, tiendas, usuarios, alertas y health del worker.")

        Container(worker, "Price Worker", "Node.js + TypeScript + node-cron", "Proceso en segundo plano. Ejecuta scrapers y llamadas API según la configuración almacenada en BD. Detecta anomalías y delega notificaciones.")

        Container(notifService, "Notification Service", "Node.js + TypeScript", "Módulo que despacha notificaciones vía Gmail SMTP (Nodemailer) o Telegram Bot API según preferencia del usuario.")

        ContainerDb(db, "Base de datos", "Supabase — PostgreSQL", "Almacena perfumes, precios históricos, tiendas, configuraciones de scraper, usuarios, favoritos y alertas.")
    }

    Rel(admin, frontend, "Accede al panel Admin", "HTTPS")
    Rel(visitor, frontend, "Navega el sitio", "HTTPS")

    Rel(frontend, api, "Consulta y mutaciones", "REST / JSON / HTTPS")

    Rel(api, db, "Lee y escribe datos", "SQL / TLS")

    Rel(worker, retailSites, "Scraping con stealth", "Puppeteer / HTTPS")
    Rel(worker, mlAPI, "Consulta precios", "REST / HTTPS")
    Rel(worker, db, "Escribe precios y logs; lee config de tiendas", "SQL / TLS")
    Rel(worker, notifService, "Delega envío de notificaciones", "Llamada interna")

    Rel(notifService, gmailSMTP, "Envía emails", "SMTP / TLS")
    Rel(notifService, telegramBot, "Envía mensajes", "HTTPS")
```

## Nivel 3 — Componentes

### 3.1 API REST

```mermaid
C4Component
    title API REST — Componentes

    Container_Ext(frontend, "Frontend", "Angular")
    Container_Ext(worker, "Price Worker")
    ContainerDb_Ext(db, "Supabase — PostgreSQL")

    Container_Boundary(api, "API REST") {

        Component(authMiddleware, "Auth Middleware", "Express Middleware", "Valida JWT emitido por Supabase Auth. Aplica control de acceso por rol (admin / visitor).")

        Component(productsController, "Products Controller", "Express Router", "GET /products — búsqueda, filtros por marca, notas olfativas, género, precio y tienda.")

        Component(pricesController, "Prices Controller", "Express Router", "GET /prices/:productId — precios actuales y serie histórica por producto y tienda.")

        Component(storesController, "Stores Controller", "Express Router", "CRUD de tiendas y configuración de scraper. Solo rol admin. Incluye endpoint de test-scrape en vivo.")

        Component(alertsController, "Alerts Controller", "Express Router", "CRUD de alertas de precio por usuario. Define umbral y canal de notificación.")

        Component(healthController, "Health Controller", "Express Router", "GET /health/worker — estado, última ejecución, errores recientes y métricas por tienda. Solo rol admin.")

        Component(usersController, "Users Controller", "Express Router", "Perfil de usuario, cambio de canal de notificación y vinculación de Telegram.")

        Component(supabaseClient, "Supabase Client", "Supabase JS SDK", "Capa de acceso a datos. Ejecuta queries sobre PostgreSQL y gestiona autenticación.")
    }

    Rel(frontend, authMiddleware, "Envía JWT en header", "HTTPS")
    Rel(authMiddleware, productsController, "Pasa request autenticado")
    Rel(authMiddleware, pricesController, "Pasa request autenticado")
    Rel(authMiddleware, storesController, "Verifica rol admin")
    Rel(authMiddleware, alertsController, "Verifica sesión")
    Rel(authMiddleware, healthController, "Verifica rol admin")
    Rel(authMiddleware, usersController, "Verifica sesión")

    Rel(productsController, supabaseClient, "Query productos")
    Rel(pricesController, supabaseClient, "Query precios e historial")
    Rel(storesController, supabaseClient, "CRUD tiendas y config scraper")
    Rel(alertsController, supabaseClient, "CRUD alertas")
    Rel(healthController, supabaseClient, "Lee logs del worker")
    Rel(usersController, supabaseClient, "Lee/escribe perfil de usuario")

    Rel(supabaseClient, db, "SQL / TLS")
    Rel(worker, healthController, "Reporta estado y errores", "REST")
```

### 3.2 Price Worker

```mermaid
C4Component
    title Price Worker — Componentes

    ContainerDb_Ext(db, "Supabase — PostgreSQL")
    System_Ext(retailSites, "Tiendas Retail")
    System_Ext(mlAPI, "MercadoLibre API")
    Container_Ext(notifService, "Notification Service")

    Container_Boundary(worker, "Price Worker") {

        Component(scheduler, "Scheduler", "node-cron", "Lee la frecuencia configurada por tienda en BD y dispara los jobs correspondientes.")

        Component(storeConfigLoader, "Store Config Loader", "Módulo TS", "Carga desde BD la configuración activa de cada tienda: URL, selectores, método (scraper/api), frecuencia.")

        Component(scraperConnector, "Scraper Connector", "Puppeteer + puppeteer-extra-plugin-stealth", "Lanza navegador headless con perfil anti-detección. Aplica los selectores CSS/XPath para extraer nombre, precio normal, precio con descuento y método de pago.")

        Component(apiConnector, "API Connector", "Axios + módulo TS", "Llama endpoints externos (MercadoLibre API). Parsea y normaliza la respuesta al mismo modelo de datos interno.")

        Component(priceNormalizer, "Price Normalizer", "Módulo TS", "Unifica el formato de precio, moneda y descuento proveniente de diferentes fuentes.")

        Component(anomalyDetector, "Anomaly Detector", "Módulo TS", "Evalúa si el resultado es válido: detecta 0 resultados, errores HTTP, precios fuera de rango estadístico. Registra el error y solicita notificación al Admin.")

        Component(alertEvaluator, "Alert Evaluator", "Módulo TS", "Compara precios nuevos contra alertas activas de usuarios. Si se supera el umbral, solicita notificación al usuario.")

        Component(priceWriter, "Price Writer", "Supabase JS SDK", "Persiste precios normalizados con timestamp en BD. Escribe logs de ejecución del worker.")
    }

    Rel(scheduler, storeConfigLoader, "Dispara job con config de tienda")
    Rel(storeConfigLoader, db, "Lee configuración de tiendas activas")

    Rel(storeConfigLoader, scraperConnector, "Si método = scraper")
    Rel(storeConfigLoader, apiConnector, "Si método = api")

    Rel(scraperConnector, retailSites, "Scraping stealth")
    Rel(apiConnector, mlAPI, "REST call")

    Rel(scraperConnector, priceNormalizer, "Datos crudos")
    Rel(apiConnector, priceNormalizer, "Datos crudos")

    Rel(priceNormalizer, anomalyDetector, "Datos normalizados")
    Rel(anomalyDetector, notifService, "Alerta al Admin si error")
    Rel(anomalyDetector, priceWriter, "Datos válidos")
    Rel(priceWriter, alertEvaluator, "Precio guardado")
    Rel(alertEvaluator, notifService, "Notifica usuario si umbral superado")
    Rel(priceWriter, db, "Escribe precios y logs")
```

### 3.3 Frontend (Angular)

```mermaid
C4Component
    title Frontend Angular — Componentes

    Container_Ext(api, "API REST")
    System_Ext(supabaseAuth, "Supabase Auth")

    Container_Boundary(frontend, "Frontend Angular") {

        Component(authModule, "Auth Module", "Angular Module", "Login, registro y gestión de sesión con Supabase Auth. Guarda JWT en memoria y lo adjunta a las llamadas a la API.")

        Component(homeModule, "Home Module", "Angular Module", "Página principal: perfumes con mayor descuento actual. Carga top descuentos desde la API.")

        Component(searchModule, "Search & Filter Module", "Angular Module", "Barra de búsqueda por nombre/marca. Filtros: marca, notas olfativas, género, precio, tienda.")

        Component(productDetailModule, "Product Detail Module", "Angular Module", "Vista de detalle de un perfume: precios actuales por tienda, precio con tarjeta/banco, mínimo histórico, ofertas activas.")

        Component(chartsModule, "Charts Module", "Angular Module + Chart.js", "Gráfico de tendencia de precio histórico. Selector de rango temporal. Comparación entre tiendas.")

        Component(favoritesModule, "Favorites Module", "Angular Module", "Lista de favoritos del usuario. Acceso rápido a productos marcados.")

        Component(alertsModule, "Alerts Module", "Angular Module", "CRUD de alertas de precio: umbral en % o precio fijo, canal de notificación (mail/Telegram/ambos).")

        Component(adminModule, "Admin Module", "Angular Module (guard: admin)", "Panel exclusivo para Administrador: CRUD de tiendas, edición de selectores, test-scrape en vivo, dashboard de salud del worker.")

        Component(profileModule, "Profile Module", "Angular Module", "Edición de perfil, cambio de contraseña, configuración de canal de notificación, vinculación de Telegram.")

        Component(apiService, "API Service", "Angular Service", "Capa centralizada de comunicación con la API REST. Gestiona JWT, interceptores HTTP y manejo de errores.")
    }

    Rel(authModule, supabaseAuth, "Login / registro / refresh token", "HTTPS")
    Rel(authModule, apiService, "Provee JWT")

    Rel(homeModule, apiService, "GET /products?sort=discount")
    Rel(searchModule, apiService, "GET /products?filters...")
    Rel(productDetailModule, apiService, "GET /prices/:productId")
    Rel(chartsModule, apiService, "GET /prices/:productId?history=true")
    Rel(favoritesModule, apiService, "GET|POST|DELETE /favorites")
    Rel(alertsModule, apiService, "GET|POST|PUT|DELETE /alerts")
    Rel(adminModule, apiService, "CRUD /stores, GET /health/worker")
    Rel(profileModule, apiService, "GET|PUT /users/me")

    Rel(apiService, api, "REST / JSON / HTTPS")
```
